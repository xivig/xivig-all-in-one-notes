     LARAVEL DYNAMIC MENU BUILDER & ADMIN CRUD (COMPLETE & UNIFIED GUIDE)

This comprehensive tutorial guides you through building a database-driven,
hierarchical Dynamic Menu Management System (Admin CRUD) in Laravel.

Features included in this guide:
- Multi-level parent/child navigation hierarchy (nested dropdowns)
- Admin CRUD interface to add, edit, sort, toggle status, and delete menus
- URL vs. Named Route support
- Target window selection (_self vs. _blank)
- Global View Composer to automatically inject active menus into layouts/navbars
- Mobile-responsive frontend Blade navigation component with Alpine.js
- AJAX drag-and-drop reordering endpoint

Run the Artisan command to create the model and migration:

Open `database/migrations/xxxx_xx_xx_xxxxxx_create_menus_table.php` and configure:

<?php

use Illuminate\Database\Migrations\Migration;
use Illuminate\Database\Schema\Blueprint;
use Illuminate\Support\Facades\Schema;

return new class extends Migration
{
    public function up(): void
    {
        Schema::create('menus', function (Blueprint $table) {
            $table->id();
            $table->string('title');                          // Menu Label (e.g., 'Home', 'Services')
            $table->string('url')->nullable();                // Custom URL (e.g., '/contact' or 'https://...')
            $table->string('route_name')->nullable();         // Named Route (e.g., 'dashboard', 'posts.index')
            $table->string('icon')->nullable();               // Icon class or SVG name (e.g., 'fa fa-home')
            $table->string('target')->default('_self');       // '_self' or '_blank'
            $table->foreignId('parent_id')                    // Parent menu for multi-level dropdowns
                  ->nullable()
                  ->constrained('menus')
                  ->cascadeOnDelete();
            $table->integer('order')->default(0);             // Display sequence
            $table->boolean('is_active')->default(true);      // Visibility toggle
            $table->string('permission')->nullable();         // Optional role/gate required
            $table->timestamps();
        });
    }

    public function down(): void
    {
        Schema::dropIfExists('menus');
    }
};

Run migration:
php artisan migrate

2. MENU ELOQUENT MODEL (RELATIONSHIPS, SCOPES & HELPERS)
================================================================================
Open `app/Models/Menu.php` and define the model:

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Support\Facades\Route;

class Menu extends Model
{
    use HasFactory;

    protected $fillable = [
        'title',
        'url',
        'route_name',
        'icon',
        'target',
        'parent_id',
        'order',
        'is_active',
        'permission',
    ];

    /**
     * Modern attribute casting.
     */
    protected function casts(): array
    {
        return [
            'is_active' => 'boolean',
            'order'     => 'integer',
        ];
    }

    /**
     * Parent menu relationship.
     */
    public function parent(): BelongsTo
    {
        return $this->belongsTo(Menu::class, 'parent_id');
    }

    /**
     * Active direct children (ordered for navigation rendering).
     */
    public function children(): HasMany
    {
        return $this->hasMany(Menu::class, 'parent_id')
                    ->where('is_active', true)
                    ->orderBy('order', 'asc');
    }

    /**
     * All children including inactive (for Admin management).
     */
    public function allChildren(): HasMany
    {
        return $this->hasMany(Menu::class, 'parent_id')
                    ->orderBy('order', 'asc');
    }

    /**
     * Scope for root menu items (top-level navbar items).
     */
    public function scopeRoot($query)
    {
        return $query->whereNull('parent_id');
    }

    /**
     * Scope for active menu items.
     */
    public function scopeActive($query)
    {
        return $query->where('is_active', true);
    }

    /**
     * Check if this menu has active child items.
     */
    public function hasChildren(): bool
    {
        return $this->children->isNotEmpty();
    }

    /**
     * Resolve the target URL (evaluates named route first, then custom url).
     */
    public function getResolvedUrl(): string
    {
        if ($this->route_name && Route::has($this->route_name)) {
            return route($this->route_name);
        }

        if ($this->url) {
            return str_starts_with($this->url, 'http') ? $this->url : url($this->url);
        }

        return '#';
    }

    /**
     * Check if the current menu corresponds to the active page.
     */
    public function isCurrent(): bool
    {
        if ($this->route_name && request()->routeIs($this->route_name)) {
            return true;
        }

        if ($this->url) {
            $trimmed = trim($this->url, '/');
            if ($trimmed !== '' && request()->is($trimmed . '*')) {
                return true;
            }
        }

        return false;
    }
}

4. ADMIN MENU CONTROLLER
================================================================================
Generate the controller:
php artisan make:controller Admin/MenuController --resource

Open `app/Http/Controllers/Admin/MenuController.php`:

<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\Menu;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\View\View;

class MenuController extends Controller
{
    /**
     * Display a listing of all menu items with hierarchy.
     */
    public function index(): View
    {
        // Load root menus with their nested children
        $menus = Menu::with(['allChildren.allChildren'])
                     ->root()
                     ->orderBy('order', 'asc')
                     ->get();

        return view('admin.menus.index', compact('menus'));
    }

    /**
     * Show the form for creating a new menu item.
     */
    public function create(): View
    {
        // Selectable parent menus (top-level only to prevent infinite nesting)
        $parentMenus = Menu::root()->orderBy('title', 'asc')->get();

        return view('admin.menus.create', compact('parentMenus'));
    }

    /**
     * Store a newly created menu item in storage.
     */
    public function store(Request $request): RedirectResponse
    {
        $validated = $request->validate([
            'title'      => ['required', 'string', 'max:255'],
            'url'        => ['nullable', 'string', 'max:255'],
            'route_name' => ['nullable', 'string', 'max:255'],
            'icon'       => ['nullable', 'string', 'max:100'],
            'target'     => ['required', 'in:_self,_blank'],
            'parent_id'  => ['nullable', 'exists:menus,id'],
            'order'      => ['nullable', 'integer', 'min:0'],
            'is_active'  => ['sometimes', 'boolean'],
        ]);

        $validated['is_active'] = $request->has('is_active');
        $validated['order'] = $validated['order'] ?? 0;

        Menu::create($validated);

        return redirect()->route('admin.menus.index')
                         ->with('success', 'Menu item created successfully.');
    }

    /**
     * Show the form for editing the specified menu item.
     */
    public function edit(Menu $menu): View
    {
        // Cannot select itself as parent
        $parentMenus = Menu::root()
                           ->where('id', '!=', $menu->id)
                           ->orderBy('title', 'asc')
                           ->get();

        return view('admin.menus.edit', compact('menu', 'parentMenus'));
    }

    /**
     * Update the specified menu item in storage.
     */
    public function update(Request $request, Menu $menu): RedirectResponse
    {
        $validated = $request->validate([
            'title'      => ['required', 'string', 'max:255'],
            'url'        => ['nullable', 'string', 'max:255'],
            'route_name' => ['nullable', 'string', 'max:255'],
            'icon'       => ['nullable', 'string', 'max:100'],
            'target'     => ['required', 'in:_self,_blank'],
            'parent_id'  => ['nullable', 'exists:menus,id', 'different:id'],
            'order'      => ['nullable', 'integer', 'min:0'],
            'is_active'  => ['sometimes', 'boolean'],
        ]);

        // Prevent setting a child as parent to avoid cyclic recursion
        if (!empty($validated['parent_id']) && $validated['parent_id'] == $menu->id) {
            return back()->withErrors(['parent_id' => 'A menu item cannot be its own parent.']);
        }

        $validated['is_active'] = $request->has('is_active');
        $validated['order'] = $validated['order'] ?? 0;

        $menu->update($validated);

        return redirect()->route('admin.menus.index')
                         ->with('success', 'Menu item updated successfully.');
    }

    /**
     * Remove the specified menu item from storage.
     */
    public function destroy(Menu $menu): RedirectResponse
    {
        $menu->delete(); // Cascade deletes child items due to DB foreign key constraint

        return redirect()->route('admin.menus.index')
                         ->with('success', 'Menu item deleted successfully.');
    }

    /**
     * Update order of menus via AJAX drag and drop.
     */
    public function reorder(Request $request)
    {
        $request->validate([
            'order' => ['required', 'array'],
            'order.*.id' => ['required', 'exists:menus,id'],
            'order.*.order' => ['required', 'integer'],
        ]);

        foreach ($request->order as $item) {
            Menu::where('id', $item['id'])->update(['order' => $item['order']]);
        }

        return response()->json(['status' => 'success', 'message' => 'Menu order updated.']);
    }
}

use App\Http\Controllers\Admin\MenuController;
use Illuminate\Support\Facades\Route;

// Protect menu management with your admin middleware:
Route::prefix('admin')->name('admin.')->middleware(['web', 'auth'])->group(function () {
    Route::post('menus/reorder', [MenuController::class, 'reorder'])->name('menus.reorder');
    Route::resource('menus', MenuController::class);
});

7. VIEW COMPOSER SETUP (SHARING MENUS GLOBALLY)
================================================================================
Rather than passing `$menus` in every controller, register a View Composer in
`app/Providers/AppServiceProvider.php` inside the `boot()` method:

use App\Models\Menu;
use Illuminate\Support\Facades\View;

public function boot(): void
{
    // Share $navigationMenus with layout and navbar views:
    View::composer(['layouts.navigation', 'layouts.app', 'partials.navbar'], function ($view) {
        $menus = Menu::with(['children.children'])
                     ->root()
                     ->active()
                     ->orderBy('order', 'asc')
                     ->get();

        $view->with('navigationMenus', $menus);
    });
}

9. DRAG-AND-DROP REORDERING (AJAX FRONTEND ENHANCEMENT)
================================================================================
To enable seamless drag-and-drop ordering in the admin index table:
Include SortableJS (via CDN in `resources/views/admin/menus/index.blade.php`):

<script src="https://cdn.jsdelivr.net/npm/sortablejs@1.15.0/Sortable.min.js"></script>
<script>
    const el = document.querySelector('tbody');
    Sortable.create(el, {
        animation: 150,
        onEnd: function () {
            let orderData = [];
            document.querySelectorAll('tbody tr').forEach((row, index) => {
                const id = row.getAttribute('data-id');
                if (id) {
                    orderData.push({ id: id, order: index + 1 });
                }
            });

            // Send AJAX update
            fetch("{{ route('admin.menus.reorder') }}", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json",
                    "X-CSRF-TOKEN": "{{ csrf_token() }}"
                },
                body: JSON.stringify({ order: orderData })
            })
            .then(res => res.json())
            .then(data => console.log('Order saved:', data));
        }
    });
</script>
